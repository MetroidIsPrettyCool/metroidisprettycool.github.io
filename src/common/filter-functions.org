  ;; YouTube embedding code adapted from
  ;; <https://endlessparentheses.com/embedding-youtube-videos-with-org-mode-links.html>

  (org-link-set-parameters
   "yt"
   :follow
   (lambda (path _uarg)
     (browse-url-xdg-open (concat "https://www.youtube.com/watch?v=" path)))

   :export
   (lambda (path desc backend _uarg)
     (pcase backend
       ('html (format
               "<iframe class=\"metroidisprettycool-github-io-yt\" src=\"https://www.youtube.com/embed/%s\" title=\"%s\"></iframe>"
               path (or desc "")))
       ('latex (let ((path (concat "https://www.youtube.com/watch?v=" path)))
                 (if desc (format "\\href{%s}{%s}" path desc) (format "\\url{%s}" path))))
       (_ path))))

  (defun metroidisprettycool-github-io--tidy-filter (string backend comm-channel)
    (when (org-export-derived-backend-p backend 'html)
      (when-let* ((stderr-tmp-file (make-temp-file "metroidisprettycool-github-io--tidy-filter"))
                  (output-file (plist-get comm-channel :output-file))
                  (input-file (plist-get comm-channel :input-file))
                  (org-project (org-publish-get-project-from-filename input-file))
                  (project-plist (cdr org-project))
                  (publishing-directory (plist-get project-plist :publishing-directory))
                  (output-filename-relative (file-relative-name output-file publishing-directory))
                  (lang (plist-get comm-channel :language)))
        (with-temp-buffer
          (insert string)
          (let ((exit-status (call-process-region nil nil "tidy" t (list t stderr-tmp-file) nil
                                                  "-language"                lang
                                                  "--add-xml-decl"           "yes"
                                                  "--output-xhtml"           "yes"
                                                  "--char-encoding"          "utf8"
                                                  "--output-bom"             "no"
                                                  "--clean"                  "yes"
                                                  "--drop-empty-elements"    "yes"
                                                  "--drop-empty-paras"       "yes"
                                                  "--merge-divs"             "yes"
                                                  "--merge-spans"            "yes"
                                                  "--numeric-entities"       "yes"
                                                  "--fix-style-tags"         "yes"
                                                  "--fix-uri"                "yes"
                                                  "--lower-literals"         "yes"
                                                  "--repeated-attributes"    "keep-last"
                                                  "--strict-tags-attributes" "yes"
                                                  "--escape-cdata"           "no"
                                                  "--join-styles"            "yes"
                                                  "--merge-emphasis"         "yes"
                                                  "--replace-color"          "yes"
                                                  "--indent"                 "yes"
                                                  "--indent-with-tabs"       "no"
                                                  "--keep-tabs"              "no"
                                                  "--tidy-mark"              "yes")))
            (with-temp-buffer
              (insert-file-contents stderr-tmp-file)
              (save-match-data
                (while (re-search-forward (rx "line" (*? space) (group-n 1 (+ digit))
                                              (*? space)
                                              "column" (*? space) (group-n 2 (+ digit))
                                              (*? space) ?- (*? space)
                                              (group-n 3 (or "Warning" "Error")) ?:
                                              (group-n 4 (*? anything))
                                              eol)
                                          nil t)
                  (message "%s:%s:%s - %s:%s"

                           output-filename-relative
                           (match-string-no-properties 1)
                           (match-string-no-properties 2)
                           (match-string-no-properties 3)
                           (match-string-no-properties 4)))))
            (delete-file stderr-tmp-file nil)
            (when (= exit-status 2) (error "Tidy exited with code indicating errors!")))
          (buffer-substring-no-properties (point-min) (point-max))))))

  (defun metroidisprettycool-github-io--normalize-urls-in-buffer (file-in-dir normalize-to-dir extension)
    "Normalizes URLs in an XHTML buffer to be relative to NORMALIZE-TO-DIR.

    Simply replaces the value of any href or src attribute that isn't a
    fully-qualified URL, doesn't have a URL scheme, doesn't have a URL path
    that already starts with /, and isn't just a bare fragment.

    Any rewritten paths that end in a / character will be suffixed
    \"index.EXTENSION\".

    In addition, when rewriting tags with href attributes, a check is
    performed to see if the tag's only content is text equal to that href
    attribute's value (e.g. <a href=\"./foo.txt\">./foo.txt</a>), optionally
    surrounded on either side with arbitrary whitespace. If it is, the
    content of the tag will be rewritten to match the href attribute value.

    Assumes buffer already contains well-formed XML; that all href and src
    attributes are valid; that FILE-IN-DIR is equal to or a subdirectory of
    NORMALIZE-TO-DIR, and that no relative path points to anything outside
    NORMALIZE-TO-DIR.

    (In other words, we're not really bothering with error-checking or
    defensive programming, besides whatever error checking we got for free
    from our library calls.)"
    (goto-char (point-min))
    (xmltok-forward-prolog)

    (while (xmltok-forward)
      (cl-loop for i upfrom 0 below (length xmltok-attributes) do
               (let* ((attr (nth i xmltok-attributes))
                      (name (xmltok-attribute-local-name attr)))
                 (when (or (string-equal name "href") (string-equal name "src"))
                   (let* ((attr-value (xmltok-attribute-value attr))
                          (url (url-generic-parse-url attr-value))
                          (filename (or (url-filename url) "")))
                     (unless (or (url-fullness url)
                                 (url-type url)
                                 (string-match-p (rx bos ?/) filename)
                                 (and (string-empty-p filename) (url-target url)))
                       (let* ((attr-start (xmltok-attribute-value-start attr))
                              (attr-end (xmltok-attribute-value-end attr))
                              (attr-len (- attr-start attr-end))
                              (expanded-filename (expand-file-name filename file-in-dir))
                              (relative-filename (concat "/" (file-relative-name expanded-filename normalize-to-dir)))
                              (new-filename (if (file-directory-p expanded-filename)
                                                (expand-file-name (concat "index." extension) relative-filename)
                                              relative-filename))
                              (new-url (url-generic-parse-url attr-value)))
                         (setf (url-filename new-url) new-filename)
                         (goto-char attr-start)
                         (delete-region attr-start attr-end)
                         (insert new-filename)
                         (goto-char xmltok-start)
                         (xmltok-forward)
                         (when (and (string-equal name "href") (eq xmltok-type 'start-tag))
                           (save-excursion
                             (xmltok-save
                               (let* ((inner-type (xmltok-forward))
                                      (inner-start xmltok-start)
                                      (inner-end (point)))
                                 (when (and (eq inner-type 'data) (eq (xmltok-forward) 'end-tag))
                                   (let* ((inner-content
                                           (buffer-substring-no-properties inner-start inner-end))
                                          (inner-len
                                           (length inner-content))

                                          (inner-content-tr-left
                                           (string-trim-left inner-content))
                                          (inner-content-tr-left-len
                                           (length inner-content-tr-left))
                                          (inner-content-left-whitespace
                                           (- inner-len inner-content-tr-left-len))

                                          (inner-content-tr-both
                                           (string-trim-right inner-content-tr-left))
                                          (inner-content-tr-both-len
                                           (length inner-content-tr-both))
                                          (inner-content-right-whitespace
                                           (- inner-content-tr-left-len inner-content-tr-both-len))

                                          (inner-text-start (+ inner-start inner-content-left-whitespace))
                                          (inner-text-end   (- inner-end   inner-content-right-whitespace)))

                                     (when (string-equal attr-value inner-content-tr-both)
                                       (goto-char     inner-text-start)
                                       (delete-region inner-text-start inner-text-end)
                                       (insert (url-recreate-url new-url)))))))))))))))))

  (defun metroidisprettycool-github-io--normalize-urls-filter (string backend comm-channel)
    (when (org-export-derived-backend-p backend 'html)
      (when-let* ((input-file (plist-get comm-channel :input-file))
                  (output-file (plist-get comm-channel :output-file))
                  (org-project (org-publish-get-project-from-filename input-file))
                  (project-plist (cdr org-project))
                  (publishing-directory (plist-get project-plist :publishing-directory))
                  (output-file-in-dir (file-name-directory output-file))
                  (extension (file-name-extension output-file)))
        (with-temp-buffer
          (insert string)
          (metroidisprettycool-github-io--normalize-urls-in-buffer output-file-in-dir
                                                                   publishing-directory
                                                                   extension)
          (buffer-substring-no-properties (point-min) (point-max))))))
